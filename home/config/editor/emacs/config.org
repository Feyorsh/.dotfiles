(defun nscolor2hex (color)
  "Converts colors from `NSColor' format to hex values"
  (concat "#"                           ; Add # at the front
          (mapconcat 'identity          ; concatenate the list
                     (mapcar #'(lambda (x) ;returns ("hex" "hex" "hex")
                                (let ((col (lsh (string-to-number x) -8)))
                                  (if (< col 16)
                                      (format "0%x" col)
                                    (format "%x" col))))
                             (split-string (s-replace "\"" "" color) ",")) "")))
(defun color-picker (&optional list buffer-name callback)
  "Calls OS X color picker and insert the chosen color."
  (interactive)
  (let ((result
         (do-applescript "tell application \"Emacs\"
set result to \"\"
set x to (choose color)
set result to item 1 of x as string
set result to result & \",\"
set result to result & item 2 of x as string
set result to result & \",\"
set result to result & item 3 of x as string
return result
end tell")))
    (if callback ; For Easy Customization
        (funcall callback (nscolor2hex result))
      (message "Copied %s" (nscolor2hex result))
      (kill-new (nscolor2hex result)))
    (do-applescript "tell application \"Emacs\" to activate")))

Below are wrappers around Applescript primitives that can be used to send native UI alerts from Emacs.
Note that macOS-level customization of these notifications (primarily the alerts) must be done through Notifications in System Settings.
The length merits refactoring into a standalone package, but I'm hesitant to do so because I know there must be an existing package out there that does the same damn thing!
#+begin_src elisp
(defun fysh/notif-notification (body &optional title subtitle sound)
  "Display a macOS notification (banner in upper right, no user interaction). Note that the display style (banner vs alert vs not displaying at all) must be configured by the user in System Settings.
See /System/Library/Sounds/ for valid sounds."
  (do-applescript
      (concat
       "display notification \"" body "\""
       (when title (concat " with title \"" title "\""))
       (when subtitle (concat " subtitle \"" subtitle "\""))
       (when sound (concat " sound name \"" sound "\"")))))

(defun fysh/notif-alert (title &optional description severity buttons default-button cancel-button timeout sound-file)
  "Display a macOS alert (centered element with buttons). Returns the string representing the selected button, or nil if the alert timed out. The argument `sound', if non-nil, will continuously loop a sound file with afplay until the alert returns."
  (let* ((alert-buf (get-buffer-create " *notif-alert*"))
         (shell-file-name "/bin/sh")
         (alert-proc (when sound-file (start-process-shell-command "notif-alert" alert-buf (concat "while :; do afplay " sound-file "; done"))))
         (ret (do-applescript
                  (concat
                   "display alert \"" title "\""
                   (when description (concat " message \"" description "\""))
                   (when severity
                     (if-let ((type (memq severity '(informational warning critical)))
                              (val (symbol-value (car type))))
                         (concat " as \"" val "\"")
                       (error (format "%s must be either informational, warning, or critical." severity))))
                   (cond ((and (or default-button cancel-button) (not buttons)) (error "Cannot set `default-buton' or `cancel-button' without `buttons'."))
                         ((and buttons (not (listp buttons))) (error "`buttons' must be a list."))
                         ((identity buttons) (if (length> buttons 3)
                                                 (error "`buttons' can have at most 3 elements.")
                                               (concat " buttons {\"" (string-join buttons "\", \"") "\"}")
                                               (when default-button
                                                 (if (not (memq default-button buttons))
                                                     (error (format "%s must be one of %s." default-button buttons))
                                                   (concat " default button \"" default-button "\"")))
                                               (when cancel-button
                                                 (cond (((string= cancel-button default-button) (error "`cancel-button' cannot be the same as `default-button'."))
                                                        ((not (memq cancel-button buttons)) (error (format "%s must be one of %s." cancel-button buttons)))
                                                        (t (concat " cancel button \"" cancel-button "\""))))))))
                   (when timeout
                     (if (integerp timeout)
                         (concat " giving up after \"" (number-to-string timeout) "\"")
                       (error (format "%s must be an integer." timeout))))))))
         (when alert-proc (set-process-query-on-exit-flag alert-proc nil))
         (when alert-buf (kill-buffer alert-buf))
         (cond ((string-match "\", gave up:true}" ret) nil)
               ((string-match "\"\\(.*\\)\"" ret) (match-string 1 ret)))))

(defun fysh/notif-dialog (text &optional default-answer hide-answer buttons default-button cancel-button title icon timeout)
  "Display a dialog (centered element with editable text and buttons). Returns the input text and the string representing the selected button, or nil if the alert timed out or the user canceled."
  (let ((ret (ignore-error error (do-applescript
                                     (concat
                                      "display dialog \"" text "\""
                                      (when default-answer (concat " default answer \"" default-answer "\""))
                                      (when (eq hide-answer t) (concat " hidden answer true")) ;; explicit check for t instead of "truthiness"
                                      (cond ((and (or default-button cancel-button) (not buttons)) (error "Cannot set `default-buton' or `cancel-button' without `buttons'."))
                                            ((and buttons (not (listp buttons))) (error "`buttons' must be a list."))
                                            ((identity buttons) (if (length> buttons 3)
                                                                    (error "`buttons' can have at most 3 elements.")
                                                                  (concat " buttons {\"" (string-join buttons "\", \"") "\"}")
                                                                  (when default-button
                                                                    (if (not (memq default-button buttons))
                                                                        (error (format "%s must be one of %s." default-button buttons))
                                                                      (concat " default button \"" default-button "\"")))
                                                                  (when cancel-button
                                                                    (cond (((string= cancel-button default-button) (error "`cancel-button' cannot be the same as `default-button'."))
                                                                           ((not (memq cancel-button buttons)) (error (format "%s must be one of %s." cancel-button buttons)))
                                                                           (t (concat " cancel button \"" cancel-button "\""))))))))
                                      (when title (concat " with title \"" title "\""))
                                      (when icon (concat " with icon \"" icon "\""))
                                      (when timeout
                                        (if (integerp timeout)
                                            (concat " giving up after \"" (number-to-string timeout) "\"")
                                          (error (format "%s must be an integer." timeout)))))))))
    (when ret
      (cond ((string-match "\", gave up:true}" ret) nil)
            ;; this is 110% vulnerable to injection; assume the worst!
            ((string-match "{button returned:\"\\(.*\\)\", text returned:\"\\(.*\\)\"}" ret) `(,(match-string 1 ret) ,(match-string 2 ret)))))))

(defun timeout--debounce-advice (&optional delay default)
  "Return a function that debounces its argument function.

DELAY defaults to 0.50 seconds.  DEFAULT is the immediate return
value of the function when called.

This is intended for use as function advice."
  (let ((debounce-timer nil)
        (delay (or delay 0.50)))
    (lambda (orig-fn &rest args)
      "Debounce calls to this function."
      (if (timerp debounce-timer)
          (timer-set-idle-time debounce-timer delay)
        (prog1 default
          (setq debounce-timer
                (run-with-idle-timer
                 delay nil
                 (lambda (buf)
                   (cancel-timer debounce-timer)
                   (setq debounce-timer nil)
                   (if (buffer-live-p buf)
                       (with-current-buffer buf
                         (apply orig-fn args))
                     (apply orig-fn args)))
                 (current-buffer))))))))

(defun timeout-debounce! (func &optional delay default)
  "Debounce FUNC by DELAY seconds.

This advises FUNC, when called (interactively or from code), to
run after DELAY seconds.   If FUNC is called again within this time,
the timer is reset.

DELAY defaults to 0.5 seconds.   Using a delay of 0 resets the
function.

DEFAULT is the immediate return value of the function when called."
  (if (and delay (= delay 0))
      (advice-remove func 'debounce)
    (advice-add func :around (timeout--debounce-advice delay default)
                '((name . debounce)
                  (depth . -99)))))

(defun org-export-ignore-headlines (data backend info)
  "Remove headlines tagged \"ignore\" retaining contents and promoting children.
Each headline tagged \"ignore\" will be removed retaining its
contents and promoting any children headlines to the level of the
parent."
  (org-element-map data 'headline
    (lambda (object)
      (when (member "ignore" (org-element-property :tags object))
        (let ((level-top (org-element-property :level object))
              level-diff)
          (mapc (lambda (el)
                  ;; recursively promote all nested headlines
                  (org-element-map el 'headline
                    (lambda (el)
                      (when (equal 'headline (org-element-type el))
                        (unless level-diff
                          (setq level-diff (- (org-element-property :level el)
                                              level-top)))
                        (org-element-put-property el
                          :level (- (org-element-property :level el)
                                    level-diff)))))
                  ;; insert back into parse tree
                  (org-element-insert-before el object))
                (org-element-contents object)))
        (org-element-extract-element object)))
    info nil)
  data)

                         (not (zerop (length file))))

                  (let* ((download-file (funcall markdown-translate-filename-function file))
                         (valid-url (ignore-errors
                                      (member (downcase (url-type (url-generic-parse-url download-file)))
                                              markdown-remote-image-protocols))))

  ;; create a diff (patchfile) for current buffer.
  ;; also useful is `diff-apply-hunk'.
  (defun fysh/create-diff (buffer dest-file)
    (interactive "bBuffer: \nFSave diff to (*.diff): ")
    (let* ((filename (file-name-nondirectory buffer-file-name))
           (diff-switches (concat "-u --label a/" filename " --label b/" filename))
           (diff-use-labels nil))
      (save-window-excursion
        (diff-buffer-with-file buffer)
        (with-current-buffer "*Diff*"
          (read-only-mode -1)
          (kill-line)
          (insert (concat "diff -u a/" filename " b/" filename))
          (re-search-forward "Diff finished." nil t)
          (forward-line -1)
          (kill-whole-line 4)
          (write-file (file-name-with-extension dest-file "diff"))
          (kill-buffer)))))
  :config
